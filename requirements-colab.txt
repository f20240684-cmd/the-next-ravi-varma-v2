# Requirements for training on Google Colab / Kaggle (CUDA GPU).
#
# Colab already ships a CUDA-enabled torch/torchvision build. We deliberately
# do NOT pin or reinstall torch here: reinstalling it is slow, can break the
# preinstalled CUDA linkage, and leaves Colab's preinstalled torchao out of
# sync with torch (the "torchao compatibility" error seen in earlier runs).
#
# xformers is intentionally NOT listed: xformers==0.0.27.post2 hard-requires
# torch==2.4.0, so installing it on Colab silently *downgrades* torch. Diffusers
# uses PyTorch's built-in scaled-dot-product attention instead, which gives
# comparable memory use on a T4 for SD1.5 at 512px. The trainer auto-detects
# xformers if you install a build matching Colab's torch yourself.
#
# Gradio / controlnet-aux / LLM clients belong to Milestones 3-4 and are not
# needed for dataset preparation, training or generation.

diffusers==0.30.3
transformers==4.44.2
peft==0.12.0
accelerate==0.34.2
safetensors==0.4.5

# bitsandbytes is intentionally NOT installed: 0.43.3 has no CUDA 12.8 binary
# and imports `triton.ops` (removed in Triton 3), and PEFT imports bitsandbytes
# whenever it is present, which crashes LoRA injection. 8-bit Adam would only
# save ~19 MB for this 3.2M-parameter LoRA, so plain AdamW is used.

pillow==10.4.0
open-clip-torch==2.26.1  # evaluation (CLIP metrics)

pyyaml==6.0.2
pydantic==2.9.1
python-dotenv==1.0.1
tqdm==4.66.5
